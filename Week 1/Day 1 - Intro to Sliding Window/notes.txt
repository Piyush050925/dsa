        arr = [-1, 2, 3, 3, 4, 5, -1], k = 4

        l = -1, r = 3
        sum = -1+2+3+3 = 7

        sum = sum - arr[l]
        l++
        l = 2, r = 4

        sum = sum + arr[r] = 6+4 = 10

        and goes on
